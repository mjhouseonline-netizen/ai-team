let isRecording = false;
let recognition = null;

function toggleVoice() {
    if (!isRecording) {
        startVoiceRecording();
    } else {
        stopVoiceRecording();
    }
}

function startVoiceRecording() {
    if (!('webkitSpeechRecognition' in window)) {
        alert('Voice input not supported in your browser');
        return;
    }
    
    recognition = new webkitSpeechRecognition();
    recognition.continuous = false;
    recognition.interimResults = false;
    
    // Show visual indicator
    showVoiceStatus('🎤 Listening...');
    isRecording = true;
    
    recognition.onresult = function(event) {
        const transcript = event.results[0][0].transcript;
        document.getElementById('messageInput').value = transcript;
        stopVoiceRecording();
    };
    
    recognition.onerror = function(event) {
        console.error('Voice recognition error:', event.error);
        stopVoiceRecording();
    };
    
    recognition.start();
}

function stopVoiceRecording() {
    if (recognition) {
        recognition.stop();
    }
    hideVoiceStatus();
    isRecording = false;
}

function showVoiceStatus(message) {
    let status = document.getElementById('voiceStatus');
    if (!status) {
        status = document.createElement('div');
        status.id = 'voiceStatus';
        status.className = 'voice-status';
        document.body.appendChild(status);
    }
    status.textContent = message;
    status.style.display = 'block';
}

function hideVoiceStatus() {
    const status = document.getElementById('voiceStatus');
    if (status) {
        status.style.display = 'none';
    }
}
```

function previewAgentAvatar(input) {
    if (input.files && input.files[0]) {
        const reader = new FileReader();
        reader.onload = function(e) {
            const preview = document.getElementById('avatarPreview');
            preview.innerHTML = `
                <img src="${e.target.result}" 
                     style="width: 100px; height: 100px; 
                            border-radius: 50%; object-fit: cover;">
            `;
        };
        reader.readAsDataURL(input.files[0]);
    }
}

function createCustomAgent() {
    const name = document.getElementById('customAgentName').value.trim();
    const description = document.getElementById('customAgentDesc').value.trim();
    const personality = document.getElementById('customAgentPersonality').value.trim();
    const avatarFile = document.getElementById('agentAvatar').files[0];
    
    if (!name || !description || !personality) {
        alert('Please fill in all required fields!');
        return;
    }
    
    // Convert image to base64 if provided
    if (avatarFile) {
        const reader = new FileReader();
        reader.onload = function(e) {
            const avatarData = e.target.result;
            sendCustomAgentData(name, description, personality, avatarData);
        };
        reader.readAsDataURL(avatarFile);
    } else {
        sendCustomAgentData(name, description, personality, null);
    }
}

function sendCustomAgentData(name, description, personality, avatar) {
    fetch('/api/custom-agent/create', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({
            name: name,
            role: description,
            instructions: personality,
            emoji: '🤖',
            avatar_image: avatar  // Base64 image data
        })
    })
    .then(response => response.json())
    .then(data => {
        if (data.success) {
            showNotification('✅ Custom agent created!');
            closeCustomAgentModal();
            setTimeout(() => location.reload(), 1000);
        }
    });
}
```
